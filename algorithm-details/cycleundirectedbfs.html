int isCycleBFS(int src, int V, int adj[][10], int deg[], int vis[]) {
    int q[100][2], head = 0, tail = 0;
    q[tail][0] = src; q[tail][1] = -1; tail++; vis[src] = 1;
    while(head < tail) {
        int node = q[head][0], parent = q[head][1]; head++;
        for(int i = 0; i < deg[node]; i++) {
            int adjNode = adj[node][i];
            if(!vis[adjNode]) {
                vis[adjNode] = 1;
                q[tail][0] = adjNode; q[tail][1] = node; tail++;
            } else if(parent != adjNode) return 1;
        }
    }
    return 0;
}
bool checkForCycle(int src, int V, vector adj[], vector& vis) {
    vis[src] = 1;
    queue> q;
    q.push({src, -1});
    while(!q.empty()) {
        int node = q.front().first, parent = q.front().second; q.pop();
        for(auto adjacentNode: adj[node]) {
            if(!vis[adjacentNode]) {
                vis[adjacentNode] = 1; q.push({adjacentNode, node});
            } else if(parent != adjacentNode) return true;
        }
    }
    return false;
}
