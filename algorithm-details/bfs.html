Breadth-First Search (BFS)

Explanation & Dry Run

Breadth-First Search starts at a source vertex and explores all neighboring nodes at the present depth before moving on to nodes at the next depth level.

Dry Run Example

Start at vertex 1. Add 1 to queue & visited array. Dequeue 1, visit neighbors 2 and 3, add them to queue. Dequeue 2, visit neighbor 4. Queue becomes [3, 4]. Level order traversal complete.

Complexity Analysis

  • Time Complexity: O(V + E) where V is vertices and E is edges.
  • Space Complexity: O(V) for visited array and BFS queue.

Real-World Applications

  • GPS & Social Media shortest distance calculations (degrees of separation).
  • Web crawling level-by-level.
  • Peer-to-peer network broadcasting.

#include <stdio.h>

void bfs(int start, int V, int adj[][10], int deg[]) {
    int vis[10] = {0}, q[10], head = 0, tail = 0;
    q[tail++] = start; vis[start] = 1;
    while (head < tail) {
        int node = q[head++];
        printf("%d ", node);
        for (int i = 0; i < deg[node]; i++) {
            int neighbor = adj[node][i];
            if (!vis[neighbor]) { vis[neighbor] = 1; q[tail++] = neighbor; }
        }
    }
}
#include <iostream>
#include <vector>
#include <queue>

std::vector<int> bfsOfGraph(int V, std::vector<int> adj[]) {
    std::vector<int> vis(V, 0), bfs;
    std::queue<int> q;
    q.push(0); vis[0] = 1;
    while(!q.empty()) {
        int node = q.front(); q.pop();
        bfs.push_back(node);
        for(auto it: adj[node]) {
            if(!vis[it]) { vis[it] = 1; q.push(it); }
        }
    }
    return bfs;
}
import java.util.*;

public class BFS {
    public ArrayList<Integer> bfsOfGraph(int V, ArrayList<ArrayList<Integer>> adj) {
        ArrayList<Integer> bfs = new ArrayList<>();
        boolean vis[] = new boolean[V];
        Queue<Integer> q = new LinkedList<>();
        q.add(0); vis[0] = true;
        while (!q.isEmpty()) {
            Integer node = q.poll(); bfs.add(node);
            for (Integer it : adj.get(node)) {
                if (!vis[it]) { vis[it] = true; q.add(it); }
            }
        }
        return bfs;
    }
}
