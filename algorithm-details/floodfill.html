void dfs(int r, int c, int image[3][3], int n, int m, int initColor, int newColor) {
    if (r < 0 || c < 0 || r >= n || c >= m || image[r][c] != initColor) return;
    image[r][c] = newColor;
    dfs(r+1, c, image, n, m, initColor, newColor);
    dfs(r-1, c, image, n, m, initColor, newColor);
    dfs(r, c+1, image, n, m, initColor, newColor);
    dfs(r, c-1, image, n, m, initColor, newColor);
}
void dfs(int r, int c, vector>& ans, vector>& image, int newColor, int delRow[], int delCol[], int iniColor) {
    ans[r][c] = newColor;
    int n = image.size(), m = image[0].size();
    for (int i = 0; i < 4; i++) {
        int nrow = r + delRow[i], ncol = c + delCol[i];
        if (nrow >= 0 && nrow < n && ncol >= 0 && ncol < m && image[nrow][ncol] == iniColor && ans[nrow][ncol] != newColor) {
            dfs(nrow, ncol, ans, image, newColor, delRow, delCol, iniColor);
        }
    }
}
public class Solution {
    public int[][] floodFill(int[][] image, int sr, int sc, int color) {
        int iniColor = image[sr][sc];
        if (iniColor != color) dfs(sr, sc, image, color, iniColor);
        return image;
    }
    private void dfs(int r, int c, int[][] image, int color, int iniColor) {
        if (r < 0 || c < 0 || r >= image.length || c >= image[0].length || image[r][c] != iniColor) return;
        image[r][c] = color;
        dfs(r+1, c, image, color, iniColor); dfs(r-1, c, image, color, iniColor);
        dfs(r, c+1, image, color, iniColor); dfs(r, c-1, image, color, iniColor);
    }
}
