// Kosaraju's Algorithm in C
void dfs1(int node, int vis[], int stack[], int *top, int adj[][10], int deg[]) {
    vis[node] = 1;
    for(int i = 0; i < deg[node]; i++) {
        int it = adj[node][i];
        if(!vis[it]) dfs1(it, vis, stack, top, adj, deg);
    }
    stack[--(*top)] = node;
}
int kosaraju(int V, vector>& adj) {
    vector vis(V, 0); stack st;
    for (int i = 0; i < V; i++) {
        if (!vis[i]) dfs(i, vis, adj, st);
    }
    vector> adjT(V);
    for (int i = 0; i < V; i++) {
        vis[i] = 0;
        for (auto it : adj[i]) adjT[it].push_back(i);
    }
    int scc = 0;
    while (!st.empty()) {
        int node = st.top(); st.pop();
        if (!vis[node]) { scc++; dfs2(node, vis, adjT); }
    }
    return scc;
}
public class Solution {
    public int kosaraju(int V, ArrayList> adj) {
        int vis[] = new int[V]; Stack st = new Stack<>();
        for (int i = 0; i < V; i++) if (vis[i] == 0) dfs(i, vis, adj, st);
        ArrayList> adjT = new ArrayList<>();
        for (int i = 0; i < V; i++) adjT.add(new ArrayList<>());
        for (int i = 0; i < V; i++) {
            vis[i] = 0;
            for (Integer it : adj.get(i)) adjT.get(it).add(i);
        }
        int scc = 0;
        while (!st.isEmpty()) {
            int node = st.pop();
            if (vis[node] == 0) { scc++; dfs2(node, vis, adjT); }
        }
        return scc;
    }
}
