Explanation & Dry Run

Bellman-Ford relaxes all E edges V-1 times. If a distance reduces on the V-th iteration, the graph contains a negative weight cycle.

Dry Run Example

Initialize dist[0]=0, others = 1e8. Iterate i = 0 to V-2: For each edge (u, v, w), if dist[u] + w < dist[v], update dist[v] = dist[u] + w. Perform 1 extra check for negative cycles.

Complexity Analysis

  • Time Complexity: O(V × E).
  • Space Complexity: O(V) for distance array.

Real-World Applications

  • Distance Vector Routing protocols (RIP).
  • Arbitrage detection in financial exchange rates.

Code Implementations (C, C++, Java)

#include <stdio.h>

void bellmanFord(int V, int E, int edges[][3], int src) {
    int dist[100];
    for (int i = 0; i < V; i++) dist[i] = 1e8;
    dist[src] = 0;
    for (int i = 0; i < V - 1; i++) {
        for (int j = 0; j < E; j++) {
            int u = edges[j][0], v = edges[j][1], w = edges[j][2];
            if (dist[u] != 1e8 && dist[u] + w < dist[v]) dist[v] = dist[u] + w;
        }
    }
}
#include <iostream>
#include <vector>

std::vector<int> bellman_ford(int V, std::vector<std::vector<int>>& edges, int S) {
    std::vector<int> dist(V, 1e8);
    dist[S] = 0;
    for (int i = 0; i < V - 1; i++) {
        for (auto it : edges) {
            int u = it[0], v = it[1], wt = it[2];
            if (dist[u] != 1e8 && dist[u] + wt < dist[v]) dist[v] = dist[u] + wt;
        }
    }
    return dist;
}
import java.util.*;

public class BellmanFord {
    static int[] bellman_ford(int V, ArrayList<ArrayList<Integer>> edges, int S) {
        int[] dist = new int[V];
        Arrays.fill(dist, (int) 1e8);
        dist[S] = 0;
        for (int i = 0; i < V - 1; i++) {
            for (ArrayList<Integer> it : edges) {
                int u = it.get(0), v = it.get(1), wt = it.get(2);
                if (dist[u] != 1e8 && dist[u] + wt < dist[v]) dist[v] = dist[u] + wt;
            }
        }
        return dist;
    }
}
