    disc[u] = low[u] = timer++;
    stack[--(*top)] = u; inStack[u] = 1;
    for(int i = 0; i < deg[u]; i++) {
        int v = adj[u][i];
        if(disc[v] == -1) {

void dfs(int u, vector& disc, vector& low, stack& st, vector& inStack, vector adj[], vector>& sccs) {
    static int timer = 0;
    disc[u] = low[u] = ++timer;
    st.push(u); inStack[u] = true;
    for (int v : adj[u]) {
        if (disc[v] == -1) {
            dfs(v, disc, low, st, inStack, adj, sccs);
            low[u] = min(low[u], low[v]);
        } else if (inStack[v]) {
            low[u] = min(low[u], disc[v]);
        }
    }
    if (low[u] == disc[u]) {
        vector component;
        while (true) {
            int v = st.top(); st.pop(); inStack[v] = false;
            component.push_back(v);
            if (u == v) break;
        }
        sccs.push_back(component);
    }
}
public class Solution {
    private int timer = 0;
    public void tarjan(int u, int[] disc, int[] low, Stack st, boolean[] inStack, ArrayList> adj) {
        disc[u] = low[u] = ++timer;
        st.push(u); inStack[u] = true;
        for (int v : adj.get(u)) {
            if (disc[v] == -1) {
                tarjan(v, disc, low, st, inStack, adj);
                low[u] = Math.min(low[u], low[v]);
            } else if (inStack[v]) {
                low[u] = Math.min(low[u], disc[v]);
            }
        }
        if (low[u] == disc[u]) {
            while (true) {
                int v = st.pop(); inStack[v] = false;
                if (u == v) break;
            }
        }
    }
