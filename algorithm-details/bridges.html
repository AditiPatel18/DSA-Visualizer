int timer = 1;
void dfsBridge(int node, int parent, int vis[], int tin[], int low[], int adj[][10], int deg[]) {
    vis[node] = 1; tin[node] = low[node] = timer++;
    for(int i = 0; i < deg[node]; i++) {
        int it = adj[node][i];
        if(it == parent) continue;
        if(!vis[it]) {
            dfsBridge(it, node, vis, tin, low, adj, deg);
            if(low[it] < low[node]) low[node] = low[it];
            if(low[it] > tin[node]) printf("Bridge: %d - %d\n", node, it);
        } else {
            if(tin[it] < low[node]) low[node] = tin[it];
        }
    }
}
int timer = 1;
void dfs(int node, int parent, vector& vis, vector& tin, vector& low, vector>& bridges, vector adj[]) {
    vis[node] = 1; tin[node] = low[node] = timer++;
    for(auto it: adj[node]) {
        if(it == parent) continue;
        if(!vis[it]) {
            dfs(it, node, vis, tin, low, bridges, adj);
            low[node] = min(low[node], low[it]);
            if(low[it] > tin[node]) bridges.push_back({node, it});
        } else {
            low[node] = min(low[node], tin[it]);
        }
    }
}
public class Solution {
    private int timer = 1;
    private void dfs(int node, int parent, int[] vis, int tin[], int low[], List> bridges, ArrayList> adj) {
        vis[node] = 1; tin[node] = low[node] = timer++;
        for (Integer it : adj.get(node)) {
            if (it == parent) continue;
            if (vis[it] == 0) {
                dfs(it, node, vis, tin, low, bridges, adj);
                low[node] = Math.min(low[node], low[it]);
                if (low[it] > tin[node]) bridges.add(Arrays.asList(node, it));
            } else {
                low[node] = Math.min(low[node], tin[it]);
            }
        }
    }
}
