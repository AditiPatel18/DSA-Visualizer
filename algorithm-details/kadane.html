3. Understanding Kadane's Algorithm

Kadane's Algorithm is an optimal dynamic programming technique that solves the Maximum Subarray Sum problem in a single pass $O(N)$ time and $O(1)$ space.

Dynamic Running Sum

Accumulate sum += arr[i]. Update maxSum = max(maxSum, sum) at each index.

Reset Negative Subarrays

If sum < 0, reset sum = 0 because a negative sum reduces subsequent totals.

6. Algorithm Steps

  1. Initialize maxSum = LONG_MIN and sum = 0.
  2. Iterate through the array from i = 0 to N-1.
  3. Add arr[i] to sum.
  4. Update maxSum = max(maxSum, sum).
  5. If sum < 0, reset sum = 0.
  6. Return maxSum after the loop completes.

8. Complexity Analysis

Best Case Time

O(N)

Single pass linear scan over array.

Average Case Time

O(N)

Evaluates each element exactly once.

Worst Case Time

O(N)

Works in linear time even for all negative numbers.

Space Complexity

O(1)

Uses auxiliary scalar counters sum and maxSum.

9. Code Implementations

C Implementation


#include <stdio.h>
#include <limits.h>

long long maxSubarraySum(int arr[], int n) {
    long long maxi = LLONG_MIN;
    long long sum = 0;
    
    for (int i = 0; i < n; i++) {
        sum += arr[i];
        if (sum > maxi) maxi = sum;
        if (sum < 0) sum = 0;
    }
    return maxi;
}

int main() {
    int arr[] = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("Maximum Subarray Sum: %lld\n", maxSubarraySum(arr, n));
    return 0;
}

#include <iostream>
#include <vector>
#include <climits>
using namespace std;

long long maxSubarraySum(const vector<int>& arr) {
    long long maxi = LONG_MIN, sum = 0;
    for (int num : arr) {
        sum += num;
        maxi = max(maxi, sum);
        if (sum < 0) sum = 0;
    }
    return maxi;
}

int main() {
    vector<int> arr = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    cout << "Maximum Subarray Sum: " << maxSubarraySum(arr) << endl;
    return 0;
}

Java Implementation


public class KadaneAlgorithm {
    public static long maxSubarraySum(int[] arr) {
        long maxi = Long.MIN_VALUE;
        long sum = 0;
        
        for (int val : arr) {
            sum += val;
            maxi = Math.max(maxi, sum);
            if (sum < 0) sum = 0;
        }
        return maxi;
    }

    public static void main(String[] args) {
        int[] arr = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
        System.out.println("Maximum Subarray Sum: " + maxSubarraySum(arr));
    }
}
