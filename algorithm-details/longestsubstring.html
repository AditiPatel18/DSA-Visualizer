Longest Substring Without Repeating Characters

Variable sliding window using left and right pointers with a hashmap storing last seen character index to skip duplicates instantly.

Implementation Code (C++)

#include <string>
#include <vector>
#include <algorithm>

int lengthOfLongestSubstring(std::string s) {
    std::vector<int> mpp(256, -1);
    int left = 0, right = 0, n = s.size(), len = 0;
    while (right < n) {
        if (mpp[s[right]] != -1) left = std::max(mpp[s[right]] + 1, left);
        mpp[s[right]] = right;
        len = std::max(len, right - left + 1);
        right++;
    }
    return len;
}
