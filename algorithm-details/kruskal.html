Explanation & Dry Run

Kruskal's algorithm sorts all graph edges by weight in ascending order. It iterates through the sorted edge list and adds each edge to the MST if its endpoints belong to different components (checked using DSU Find-Parent), thereby avoiding cycles.

Dry Run Example

Sorted edges: (1-2, w=1), (2-3, w=2), (1-3, w=3).
1. Add (1-2, w=1) → DSU Union(1, 2).
2. Add (2-3, w=2) → DSU Union(2, 3).
3. Check (1-3, w=3) → Find(1) == Find(3), skip to prevent cycle.

Complexity Analysis

  • Time Complexity: O(E log E) for sorting edges.
  • Space Complexity: O(V + E) for DSU structures and edge list.

Real-World Applications

  • LAN network cable infrastructure wiring.
  • Image segmentation and cluster analysis.

Code Implementations (C, C++, Java)

#include <stdio.h>
#include <stdlib.h>

struct Edge { int u, v, w; };
int parent[100];

int findUPar(int node) {
    if (node == parent[node]) return node;
    return parent[node] = findUPar(parent[node]);
}
int cmp(const void* a, const void* b) {
    return ((struct Edge*)a)->w - ((struct Edge*)b)->w;
}
void kruskal(struct Edge edges[], int V, int E) {
    qsort(edges, E, sizeof(struct Edge), cmp);
    for (int i = 0; i < V; i++) parent[i] = i;
    int mstWt = 0;
    for (int i = 0; i < E; i++) {
        int u = findUPar(edges[i].u), v = findUPar(edges[i].v);
        if (u != v) { parent[u] = v; mstWt += edges[i].w; }
    }
}
#include <iostream>
#include <vector>
#include <algorithm>

struct DisjointSet {
    std::vector<int> parent;
    DisjointSet(int n) { parent.resize(n); for(int i=0;i<n;i++) parent[i]=i; }
    int findUPar(int node) { return parent[node] == node ? node : parent[node] = findUPar(parent[node]); }
    void unionByNode(int u, int v) { int ulp_u = findUPar(u), ulp_v = findUPar(v); if(ulp_u != ulp_v) parent[ulp_u] = ulp_v; }
};

int kruskal(int V, std::vector<std::vector<int>>& edges) {
    std::sort(edges.begin(), edges.end(), [](const auto& a, const auto& b){ return a[2] < b[2]; });
    DisjointSet ds(V); int mstWt = 0;
    for (auto it : edges) {
        if (ds.findUPar(it[0]) != ds.findUPar(it[1])) {
            mstWt += it[2]; ds.unionByNode(it[0], it[1]);
        }
    }
    return mstWt;
}
import java.util.*;

public class Kruskal {
    static int kruskal(int V, int[][] edges) {
        Arrays.sort(edges, (a, b) -> a[2] - b[2]);
        int[] parent = new int[V];
        for (int i = 0; i < V; i++) parent[i] = i;
        int mstWt = 0;
        for (int[] edge : edges) {
            int u = find(parent, edge[0]), v = find(parent, edge[1]);
            if (u != v) { parent[u] = v; mstWt += edge[2]; }
        }
        return mstWt;
    }
    static int find(int[] parent, int i) {
        if (parent[i] == i) return i;
        return parent[i] = find(parent, parent[i]);
    }
}
