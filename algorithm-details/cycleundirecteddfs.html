int dfsCycle(int node, int parent, int adj[][10], int deg[], int vis[]) {
    vis[node] = 1;
    for(int i = 0; i < deg[node]; i++) {
        int adjNode = adj[node][i];
        if(!vis[adjNode]) {
            if(dfsCycle(adjNode, node, adj, deg, vis)) return 1;
        } else if(adjNode != parent) return 1;
    }
    return 0;
}
bool dfs(int node, int parent, vector adj[], vector& vis) {
    vis[node] = 1;
    for(auto adjacentNode: adj[node]) {
        if(!vis[adjacentNode]) {
            if(dfs(adjacentNode, node, adj, vis)) return true;
        } else if(adjacentNode != parent) return true;
    }
    return false;
}
public class Solution {
    private boolean dfs(int node, int parent, boolean vis[], ArrayList> adj) {
        vis[node] = true;
        for(int adjacentNode: adj.get(node)) {
            if(!vis[adjacentNode]) {
                if(dfs(adjacentNode, node, vis, adj)) return true;
            } else if(adjacentNode != parent) return true;
        }
        return false;
    }
}
