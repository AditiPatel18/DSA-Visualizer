Understanding Quick Sort

QuickSort is a Divide and Conquer algorithm. It picks an element as pivot and partitions the given array around the picked pivot. Different versions of QuickSort pick pivot in different ways:

Always pick first element as pivot

Simple but leads to O(n²) for sorted arrays

Always pick last element as pivot

Common implementation (shown in examples)

Pick a random element as pivot

Reduces chance of worst case O(n²)

Pick median as pivot

Optimal for performance, avoids worst case

Algorithm Steps

  1. Choose Pivot: Select an element from the array as the pivot
  2. Partition: Rearrange array so elements less than pivot come before, and elements greater than pivot come after
  3. Recurse: Apply quick sort recursively to left and right sub-arrays
  4. Combine: Since the sub-arrays are sorted in place, no combine step needed

C Implementation

35 lines 

#include <stdio.h>

// Function to swap two elements
void swap(int* a, int* b) {
    int t = *a;
    *a = *b;
    *b = t;
}

// Partition function - takes last element as pivot
int partition(int arr[], int low, int high) {
    int pivot = arr[high];  // pivot element
    int i = (low - 1);      // index of smaller element
    
    for (int j = low; j <= high - 1; j++) {
        // If current element is smaller than pivot
        if (arr[j] < pivot) {
            i++;  // increment index of smaller element
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high]);
    return (i + 1);
}

// QuickSort function
void quickSort(int arr[], int low, int high) {
    if (low < high) {
        // pi is partitioning index
        int pi = partition(arr, low, high);
        
        // Separately sort elements before and after partition
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

// Function to print array
void printArray(int arr[], int size) {
    for (int i = 0; i < size; i++)
        printf("%d ", arr[i]);
    printf("\\n");
}

// Main function - Driver code
int main() {
    int arr[] = {10, 7, 8, 9, 1, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    printf("Original array: ");
    printArray(arr, n);
    
    quickSort(arr, 0, n - 1);
    
    printf("Sorted array: ");
    printArray(arr, n);
    return 0;
}

Key Points - C Implementation:

  • Uses pointers for swap function
  • Last element is chosen as pivot
  • In-place sorting (O(1) extra space)
  • Time complexity: O(n log n) average case
  • Space complexity: O(log n) for recursion stack

C++ Implementation

#include <iostream>
#include <vector>
using namespace std;

class QuickSort {
private:
    // Partition function
    int partition(vector<int>& arr, int low, int high) {
        int pivot = arr[high];  // pivot element
        int i = low - 1;        // index of smaller element
        
        for (int j = low; j < high; j++) {
            // If current element is smaller than pivot
            if (arr[j] < pivot) {
                i++;
                swap(arr[i], arr[j]);
            }
        }
        swap(arr[i + 1], arr[high]);
        return i + 1;
    }
    
public:
    // QuickSort function (public interface)
    void sort(vector<int>& arr) {
        quickSort(arr, 0, arr.size() - 1);
    }
    
private:
    // Recursive QuickSort
    void quickSort(vector<int>& arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }
    
public:
    // Utility function to print array
    void printArray(const vector<int>& arr) {
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
    }
};

// Main function
int main() {
    QuickSort sorter;
    vector<int> arr = {10, 7, 8, 9, 1, 5};
    
    cout << "Original array: ";
    sorter.printArray(arr);
    
    sorter.sort(arr);
    
    cout << "Sorted array: ";
    sorter.printArray(arr);
    
    return 0;
}

Key Points - C++ Implementation:

  • Uses vector instead of raw arrays
  • Object-oriented design with class
  • std::swap function for swapping
  • Encapsulated implementation
  • Better type safety compared to C

Java Implementation

45 lines 

public class QuickSort {
    
    // Method to partition the array
    private int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;  // index of smaller element
        
        for (int j = low; j < high; j++) {
            // If current element is smaller than pivot
            if (arr[j] < pivot) {
                i++;
                
                // Swap arr[i] and arr[j]
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        
        // Swap arr[i+1] and arr[high] (or pivot)
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        
        return i + 1;
    }
    
    // QuickSort method
    public void quickSort(int[] arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }
    
    // Utility method to print array
    public void printArray(int[] arr) {
        for (int value : arr) {
            System.out.print(value + " ");
        }
        System.out.println();
    }
    
    // Driver code
    public static void main(String[] args) {
        QuickSort sorter = new QuickSort();
        int[] arr = {10, 7, 8, 9, 1, 5};
        
        System.out.print("Original array: ");
        sorter.printArray(arr);
        
        sorter.quickSort(arr, 0, arr.length - 1);
        
        System.out.print("Sorted array: ");
        sorter.printArray(arr);
    }
}

Key Points - Java Implementation:

  • Class-based implementation
  • Uses array indexing similar to C
  • No pointers - uses array indices
  • Platform independent
  • Automatic memory management

Complexity Analysis

Time Complexity

Best Case: O(n log n)

When partition always divides array into two equal halves

Average Case: O(n log n)

Random pivot selection gives this complexity on average

Worst Case: O(n²)

When array is already sorted and pivot is always smallest/largest

Space Complexity

Auxiliary Space: O(log n)

Due to recursive call stack in average case

Worst Case Space: O(n)

In worst case (sorted array), recursion depth is n

In-place: Yes

Only requires constant extra space for swapping
