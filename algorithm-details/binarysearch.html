Understanding Binary Search

Binary Search is an efficient algorithm for finding an item from a sorted list of items. It works by repeatedly dividing the search interval in half. If the value of the search key is less than the item in the middle of the interval, narrow the interval to the lower half. Otherwise, narrow it to the upper half.

Algorithm Steps

  1. Initialize: Set start = 0, end = n-1 where n is array size
  2. Find Middle: mid = start + (end - start) / 2
  3. Compare: If arr[mid] == target, return mid
  4. Adjust Left: If arr[mid] > target, set end = mid - 1
  5. Adjust Right: If arr[mid] < target, set start = mid + 1
  6. Repeat: Continue until start > end
  7. Return: If not found, return -1

Visual Example

Searching for 42 in sorted array:

Step 1: Check middle (40) → 42 > 40 → search right half
Step 2: Check middle (60) → 42 < 60 → search left half
Step 3: Check 42 → found at position 4!

Implementation in Multiple Languages

C Implementation

20 lines 

#include <stdio.h>

// Iterative Binary Search
int binarySearch(int arr[], int left, int right, int target) {
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        // Check if target is present at mid
        if (arr[mid] == target)
            return mid;
        
        // If target greater, ignore left half
        if (arr[mid] < target)
            left = mid + 1;
        // If target smaller, ignore right half
        else
            right = mid - 1;
    }
    
    // Target not present
    return -1;
}

// Recursive Binary Search
int binarySearchRecursive(int arr[], int left, int right, int target) {
    if (right >= left) {
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target)
            return mid;
        
        if (arr[mid] > target)
            return binarySearchRecursive(arr, left, mid - 1, target);
        
        return binarySearchRecursive(arr, mid + 1, right, target);
    }
    
    return -1;
}

int main() {
    int arr[] = {2, 3, 4, 10, 40};
    int n = sizeof(arr) / sizeof(arr[0]);
    int target = 10;
    
    int result = binarySearch(arr, 0, n - 1, target);
    
    if (result == -1)
        printf("Element not found\n");
    else
        printf("Element found at index %d\n", result);
    
    return 0;
}

#include <iostream>
#include <vector>
using namespace std;

class BinarySearch {
public:
    // Iterative implementation
    int searchIterative(const vector<int>& arr, int target) {
        int left = 0;
        int right = arr.size() - 1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target)
                return mid;
            
            if (arr[mid] < target)
                left = mid + 1;
            else
                right = mid - 1;
        }
        
        return -1;
    }
    
    // Recursive implementation
    int searchRecursive(const vector<int>& arr, int left, int right, int target) {
        if (right >= left) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target)
                return mid;
            
            if (arr[mid] > target)
                return searchRecursive(arr, left, mid - 1, target);
            
            return searchRecursive(arr, mid + 1, right, target);
        }
        
        return -1;
    }
};

int main() {
    BinarySearch bs;
    vector<int> arr = {2, 3, 4, 10, 40};
    int target = 10;
    
    int result = bs.searchIterative(arr, target);
    
    if (result == -1)
        cout << "Element not found" << endl;
    else
        cout << "Element found at index " << result << endl;
    
    return 0;
}

                right = mid - 1;
        }
        
        return -1;
    }
    
    // Recursive binary search
    public static int searchRecursive(int[] arr, int left, int right, int target) {
        if (right >= left) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target)
                return mid;

Complexity Analysis

Time Complexity

Best Case: O(1)

Target is at the middle of the array

Average Case: O(log n)

Target is anywhere in the array

Worst Case: O(log n)

Target is at the beginning or end, or not present

Space Complexity

Iterative: O(1)

Only uses constant extra space

Recursive: O(log n)

Due to recursion call stack

Comparison with Linear Search

Aspect Binary Search Linear Search
Time (Sorted) O(log n) O(n)
Time (Unsorted) Not applicable O(n)
Space O(1) O(1)
Implementation Medium Easy
