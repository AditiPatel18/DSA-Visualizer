Understanding Next Greater Element

Find the first element to the right of arr[i] that is strictly greater than arr[i] using a decreasing monotonic stack from right to left.

Implementation Code

void nextGreaterElement(int arr[], int n, int nge[]) {
    int st[100], top = -1;
    for (int i = n - 1; i >= 0; i--) {
        while (top >= 0 && st[top] <= arr[i]) top--;
        nge[i] = (top == -1) ? -1 : st[top];
        st[++top] = arr[i];
    }
}
#include <vector>
#include <stack>

std::vector<int> nextGreaterElement(std::vector<int>& arr) {
    int n = arr.size();
    std::vector<int> nge(n, -1);
    std::stack<int> st;
    for (int i = n - 1; i >= 0; i--) {
        while (!st.empty() && st.top() <= arr[i]) st.pop();
        if (!st.empty()) nge[i] = st.top();
        st.push(arr[i]);
    }
    return nge;
}
import java.util.*;

public class NGE {
    public static int[] nextGreater(int[] arr) {
        int n = arr.length;
        int[] nge = new int[n];
        Stack<Integer> st = new Stack<>();
        for (int i = n - 1; i >= 0; i--) {
            while (!st.isEmpty() && st.peek() <= arr[i]) st.pop();
            nge[i] = st.isEmpty() ? -1 : st.peek();
            st.push(arr[i]);
        }
        return nge;
    }
}

Complexity Analysis

  • Time Complexity: O(N) since each element is pushed and popped at most once.
  • Space Complexity: O(N) for stack and result array.
