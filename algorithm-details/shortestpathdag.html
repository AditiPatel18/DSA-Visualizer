void topoSort(int node, vector> adj[], vector& vis, stack& st) {
    vis[node] = 1;
    for (auto it : adj[node]) {
        if (!vis[it.first]) topoSort(it.first, adj, vis, st);
    }
    st.push(node);
}
vector shortestPath(int N, int M, vector>& edges) {
    vector> adj[N];
    for (int i = 0; i < M; i++) adj[edges[i][0]].push_back({edges[i][1], edges[i][2]});
    vector vis(N, 0); stack st;
    for (int i = 0; i < N; i++) if (!vis[i]) topoSort(i, adj, vis, st);
    vector dist(N, 1e9); dist[0] = 0;
    while (!st.empty()) {
        int node = st.top(); st.pop();
        if (dist[node] != 1e9) {
            for (auto it : adj[node]) {
                if (dist[node] + it.second < dist[it.first]) dist[it.first] = dist[node] + it.second;
            }
        }
    }
    return dist;
}
public class Solution {
    public int[] shortestPath(int N, int M, int[][] edges) {
        ArrayList> adj = new ArrayList<>();
        for (int i = 0; i < N; i++) adj.add(new ArrayList<>());
        for (int i = 0; i < M; i++) adj.get(edges[i][0]).add(new Pair(edges[i][1], edges[i][2]));
        int vis[] = new int[N]; Stack st = new Stack<>();
        for (int i = 0; i < N; i++) if (vis[i] == 0) topoSort(i, adj, vis, st);
        int dist[] = new int[N]; Arrays.fill(dist, (int)1e9); dist[0] = 0;
        while (!st.isEmpty()) {
            int node = st.pop();
            if (dist[node] != 1e9) {
                for (Pair it : adj.get(node)) {
                    if (dist[node] + it.weight < dist[it.node]) dist[it.node] = dist[node] + it.weight;
                }
            }
        }
        return dist;
    }
