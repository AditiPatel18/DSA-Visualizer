Explanation & Dry Run

Dijkstra's algorithm maintains a distance array initialized to infinity and source to 0. It repeatedly selects the unvisited node with minimum distance using a Min-Heap / Priority Queue and relaxes its adjacent edges.

Dry Run Example

Source node 0: dist[0] = 0, others = ∞. Extract 0, relax edge (0-1, w=4) → dist[1]=4, (0-2, w=2) → dist[2]=2. Next extract 2, relax edge (2-1, w=1) → dist[1]=min(4, 2+1)=3.

Complexity Analysis

  • Time Complexity: O(E log V) with Min-Heap.
  • Space Complexity: O(V) for distance array and Min-Heap.

Real-World Applications

  • Google Maps and OpenStreetMap route navigation.
  • Network Routing Protocols (OSPF, IS-IS).

Code Implementations (C, C++, Java)

#include <stdio.h>

void dijkstra(int graph[5][5], int V, int src) {
    int dist[5], vis[5] = {0};
    for (int i = 0; i < V; i++) dist[i] = 1e9;
    dist[src] = 0;
    for (int count = 0; count < V - 1; count++) {
        int minD = 1e9, u = -1;
        for (int v = 0; v < V; v++)
            if (!vis[v] && dist[v] <= minD) { minD = dist[v]; u = v; }
        vis[u] = 1;
        for (int v = 0; v < V; v++)
            if (!vis[v] && graph[u][v] && dist[u] + graph[u][v] < dist[v])
                dist[v] = dist[u] + graph[u][v];
    }
}
#include <iostream>
#include <vector>
#include <queue>

std::vector<int> dijkstra(int V, std::vector<std::vector<int>> adj[], int S) {
    std::priority_queue<std::pair<int, int>, std::vector<std::pair<int, int>>, std::greater<std::pair<int, int>>> pq;
    std::vector<int> dist(V, 1e9);
    dist[S] = 0; pq.push({0, S});
    while (!pq.empty()) {
        int dis = pq.top().first, node = pq.top().second; pq.pop();
        for (auto it : adj[node]) {
            int adjNode = it[0], edgeWeight = it[1];
            if (dis + edgeWeight < dist[adjNode]) {
                dist[adjNode] = dis + edgeWeight;
                pq.push({dist[adjNode], adjNode});
            }
        }
    }
    return dist;
}
import java.util.*;

public class Dijkstra {
    static int[] dijkstra(int V, ArrayList<ArrayList<ArrayList<Integer>>> adj, int S) {
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        int[] dist = new int[V];
        Arrays.fill(dist, (int) 1e9);
        dist[S] = 0; pq.add(new int[]{0, S});
        while (!pq.isEmpty()) {
            int[] cell = pq.poll();
            int dis = cell[0], node = cell[1];
            for (ArrayList<Integer> it : adj.get(node)) {
                int adjNode = it.get(0), weight = it.get(1);
                if (dis + weight < dist[adjNode]) {
                    dist[adjNode] = dis + weight;
                    pq.add(new int[]{dist[adjNode], adjNode});
                }
            }
        }
        return dist;
    }
}
