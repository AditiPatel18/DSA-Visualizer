Understanding Monotonic Stack

A Monotonic Stack maintains its elements in strictly increasing or strictly decreasing order. It solves range minimum/maximum and nearest smaller/greater element problems in O(N) linear time.

Implementation Code

// Monotonic Increasing Stack in C
void monotonicIncStack(int arr[], int n) {
    int st[100], top = -1;
    for (int i = 0; i < n; i++) {
        while (top >= 0 && st[top] > arr[i]) top--;
        st[++top] = arr[i];
    }
}
#include <vector>
#include <stack>

void monotonicIncStack(std::vector<int>& arr) {
    std::stack<int> st;
    for (int x : arr) {
        while (!st.empty() && st.top() > x) st.pop();
        st.push(x);
    }
}
import java.util.Stack;

public class MonotonicStack {
    public static void monotonicIncStack(int[] arr) {
        Stack<Integer> st = new Stack<>();
        for (int x : arr) {
            while (!st.isEmpty() && st.peek() > x) st.pop();
            st.push(x);
        }
    }
}
