int timer = 1;
void dfsAP(int node, int parent, int vis[], int tin[], int low[], int mark[], int adj[][10], int deg[]) {
    vis[node] = 1; tin[node] = low[node] = timer++;
    int child = 0;
    for(int i = 0; i < deg[node]; i++) {
        int it = adj[node][i];
        if(it == parent) continue;
        if(!vis[it]) {
            dfsAP(it, node, vis, tin, low, mark, adj, deg);
            if(low[it] < low[node]) low[node] = low[it];
            if(low[it] >= tin[node] && parent != -1) mark[node] = 1;
            child++;
        } else {
            if(tin[it] < low[node]) low[node] = tin[it];
        }
    }
    if(child > 1 && parent == -1) mark[node] = 1;
}
int timer = 1;
void dfs(int node, int parent, vector& vis, int tin[], int low[], vector& mark, vector adj[]) {
    vis[node] = 1; tin[node] = low[node] = timer++; int child = 0;
    for(auto it: adj[node]) {
        if(it == parent) continue;
        if(!vis[it]) {
            dfs(it, node, vis, tin, low, mark, adj);
            low[node] = min(low[node], low[it]);
            if(low[it] >= tin[node] && parent != -1) mark[node] = 1;
            child++;
        } else {
            low[node] = min(low[node], tin[it]);
        }
    }
    if(child > 1 && parent == -1) mark[node] = 1;
}
public class Solution {
    private int timer = 1;
    private void dfs(int node, int parent, int[] vis, int[] tin, int[] low, int[] mark, ArrayList> adj) {
        vis[node] = 1; tin[node] = low[node] = timer++; int child = 0;
        for (Integer it : adj.get(node)) {
            if (it == parent) continue;
            if (vis[it] == 0) {
                dfs(it, node, vis, tin, low, mark, adj);
                low[node] = Math.min(low[node], low[it]);
                if (low[it] >= tin[node] && parent != -1) mark[node] = 1;
                child++;
            } else {
                low[node] = Math.min(low[node], tin[it]);
            }
        }
        if (child > 1 && parent == -1) mark[node] = 1;
    }
}
