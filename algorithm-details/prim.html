Explanation & Dry Run

Prim's algorithm starts at node 0 and maintains a visited array and a priority queue storing (weight, node, parent) triplets. At each step it picks the cheapest edge connecting an unvisited vertex to the growing MST.

Dry Run Example

Start at 0: Push (wt=0, node=0, parent=-1) into PQ. Pop 0, mark visited. Push adjacent edges (w=2, node=1, parent=0) and (w=1, node=2, parent=0) into PQ. Pop smallest edge weight 1 (node 2). Add edge (0-2) to MST.

Complexity Analysis

  • Time Complexity: O(E log V) with Priority Queue.
  • Space Complexity: O(V + E) for priority queue and visited array.

Real-World Applications

  • Telecommunications & Fiber optic cable layout optimization.
  • Electrical grid network design with minimum cost.

Code Implementations (C, C++, Java)

#include <stdio.h>

void primMST(int graph[5][5], int V) {
    int key[5], vis[5] = {0}, parent[5];
    for (int i = 0; i < V; i++) key[i] = 1e9;
    key[0] = 0; parent[0] = -1;
    for (int count = 0; count < V - 1; count++) {
        int minK = 1e9, u = -1;
        for (int v = 0; v < V; v++)
            if (!vis[v] && key[v] < minK) { minK = key[v]; u = v; }
        vis[u] = 1;
        for (int v = 0; v < V; v++)
            if (graph[u][v] && !vis[v] && graph[u][v] < key[v]) {
                parent[v] = u; key[v] = graph[u][v];
            }
    }
}
#include <iostream>
#include <vector>
#include <queue>

int spanningTree(int V, std::vector<std::vector<int>> adj[]) {
    std::priority_queue<std::pair<int, int>, std::vector<std::pair<int, int>>, std::greater<std::pair<int, int>>> pq;
    std::vector<int> vis(V, 0);
    pq.push({0, 0});
    int sum = 0;
    while (!pq.empty()) {
        auto it = pq.top(); pq.pop();
        int wt = it.first, node = it.second;
        if (vis[node]) continue;
        vis[node] = 1; sum += wt;
        for (auto ed : adj[node]) {
            int adjNode = ed[0], edW = ed[1];
            if (!vis[adjNode]) pq.push({edW, adjNode});
        }
    }
    return sum;
}
import java.util.*;

public class PrimMST {
    static int spanningTree(int V, ArrayList<ArrayList<ArrayList<Integer>>> adj) {
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        boolean[] vis = new boolean[V];
        pq.add(new int[]{0, 0});
        int sum = 0;
        while (!pq.isEmpty()) {
            int[] cell = pq.poll();
            int wt = cell[0], node = cell[1];
            if (vis[node]) continue;
            vis[node] = true; sum += wt;
            for (ArrayList<Integer> ed : adj.get(node)) {
                int adjNode = ed.get(0), edW = ed.get(1);
                if (!vis[adjNode]) pq.add(new int[]{edW, adjNode});
            }
        }
        return sum;
    }
}
